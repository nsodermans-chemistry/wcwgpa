  <script>
    // Point values for grade scales
    const REGULAR_POINTS = { "A": 4, "B": 3, "C": 2, "D": 1, "F": 0 };
    const HONORS_POINTS  = { "A": 5, "B": 4, "C": 3, "D": 2, "F": 0 };
    const SOLANO_POINTS  = { "A": 5, "B": 4, "C": 3, "D": 2, "F": 0 };

    // Unit Weights
    const REGULAR_UNITS = 5;
    const HONORS_UNITS  = 5;
    const SOLANO_UNITS  = 10;

    let animationFrameId = null;
    let sparkles = [];

    // Inject SVG logo image URL
    window.addEventListener("DOMContentLoaded", function() {
      const svgLogo = `data:image/svg+xml;utf8,<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"><path fill="%23F59E0B" stroke="%231E3A8A" stroke-width="4" d="M20,20 Q50,5 80,20 Q95,50 80,80 Q50,95 20,80 Q5,50 20,20 Z"/><text x="50" y="62" font-family="sans-serif" font-weight="900" font-size="38" fill="%231E3A8A" text-anchor="middle">W</text></svg>`;
      document.getElementById("wildcatLogo").src = svgLogo;

      populateDropdownOptions("regularCount");
      populateDropdownOptions("solanoCount");
      populateDropdownOptions("honorsCount");
      initCanvas();
    });

    function populateDropdownOptions(id) {
      const select = document.getElementById(id);
      for (let i = 0; i <= 10; i++) {
        const opt = document.createElement("option");
        opt.value = i;
        opt.textContent = i === 0 ? "0 Classes" : (i === 1 ? "1 Class" : `${i} Classes`);
        select.appendChild(opt);
      }
    }

    function generateCourseRows() {
      const regNum = parseInt(document.getElementById("regularCount").value) || 0;
      const solNum = parseInt(document.getElementById("solanoCount").value) || 0;
      const honNum = parseInt(document.getElementById("honorsCount").value) || 0;

      toggleSection("regularSection", "regularRows", regNum, "Regular", "regularCountBadge");
      toggleSection("solanoSection", "solanoRows", solNum, "Solano CC", "solanoCountBadge");
      toggleSection("honorsSection", "honorsRows", honNum, "Honors", "honorsCountBadge");

      const totalCount = regNum + honNum + solNum;
      document.getElementById("emptyPrompt").style.display = totalCount === 0 ? "block" : "none";
      document.getElementById("actionArea").style.display = totalCount > 0 ? "block" : "none";
      
      document.getElementById("resultsSection").classList.add("hidden");
      stopSparkles();
    }

    function toggleSection(sectionId, containerId, count, labelPrefix, badgeId) {
      const section = document.getElementById(sectionId);
      const container = document.getElementById(containerId);
      const badge = document.getElementById(badgeId);

      if (count === 0) {
        section.classList.add("hidden");
        container.innerHTML = "";
        return;
      }

      section.classList.remove("hidden");
      badge.textContent = `${count} ${count === 1 ? 'Class' : 'Classes'}`;

      const existingGrades = Array.from(container.querySelectorAll('.course-grade')).map(select => select.value);
      container.innerHTML = "";

      for (let i = 1; i <= count; i++) {
        const row = document.createElement("div");
        row.className = "flex items-center justify-between gap-3 bg-slate-50 p-3 rounded-xl border border-slate-200 animate-fade-in";
        
        const defaultGrade = existingGrades[i-1] || "A";

        row.innerHTML = `
          <span class="text-sm font-semibold text-slate-700 whitespace-nowrap">${labelPrefix} #${i}</span>
          <select class="course-grade bg-white border border-slate-300 rounded-lg px-3 py-1.5 text-sm font-semibold text-slate-800 focus:outline-none focus:ring-2 focus:ring-blue-600">
            <option value="A" ${defaultGrade === "A" ? "selected" : ""}>A</option>
            <option value="B" ${defaultGrade === "B" ? "selected" : ""}>B</option>
            <option value="C" ${defaultGrade === "C" ? "selected" : ""}>C</option>
            <option value="D" ${defaultGrade === "D" ? "selected" : ""}>D</option>
            <option value="F" ${defaultGrade === "F" ? "selected" : ""}>F</option>
          </select>
        `;
        container.appendChild(row);
      }
    }

    function calculateGPA() {
      let totalUnits = 0;
      let totalWeightedPoints = 0;

      // Regular Classes
      const regGrades = document.querySelectorAll("#regularRows .course-grade");
      regGrades.forEach(select => {
        const pts = REGULAR_POINTS[select.value];
        totalWeightedPoints += pts * REGULAR_UNITS;
        totalUnits += REGULAR_UNITS;
      });

      // Solano Classes
      const solGrades = document.querySelectorAll("#solanoRows .course-grade");
      solGrades.forEach(select => {
        const pts = SOLANO_POINTS[select.value];
        totalWeightedPoints += pts * SOLANO_UNITS;
        totalUnits += SOLANO_UNITS;
      });

      // Honors Classes
      const honGrades = document.querySelectorAll("#honorsRows .course-grade");
      honGrades.forEach(select => {
        const pts = HONORS_POINTS[select.value];
        totalWeightedPoints += pts * HONORS_UNITS;
        totalUnits += HONORS_UNITS;
      });

      if (totalUnits === 0) return;

      const gpa = totalWeightedPoints / totalUnits;
      displayResults(gpa, totalUnits, totalWeightedPoints, regGrades.length, solGrades.length, honGrades.length);
    }

    function displayResults(gpa, units, points, regCount, solCount, honCount) {
      const resultsSection = document.getElementById("resultsSection");
      const gpaValueEl = document.getElementById("gpaValue");

      resultsSection.classList.remove("hidden");

      const formattedGpa = gpa.toFixed(2);
      gpaValueEl.textContent = formattedGpa;

      document.getElementById("breakdownUnits").textContent = units;
      document.getElementById("breakdownPoints").textContent = points;
      document.getElementById("breakdownCourses").textContent = regCount + solCount + honCount;

      const colorStyle = getGPAColor(gpa);
      gpaValueEl.style.color = colorStyle.color;
      gpaValueEl.style.textShadow = colorStyle.shadow;

      if (gpa > 4.0) {
        startSparkles(gpa);
      } else {
        stopSparkles();
      }

      resultsSection.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
    }

    function getGPAColor(gpa) {
      if (gpa > 4.0) {
        return {
          color: "#059669", 
          shadow: "0 0 20px rgba(16, 185, 129, 0.4)"
        };
      }

      const ratio = Math.max(0, Math.min(1, gpa / 4.0));
      let r, g, b;

      if (ratio < 0.33) {
        const factor = ratio / 0.33;
        r = Math.round(225 + (234 - 225) * factor);
        g = Math.round(29 + (88 - 29) * factor);
        b = Math.round(72 + (12 - 72) * factor);
      } else if (ratio < 0.66) {
        const factor = (ratio - 0.33) / 0.33;
        r = Math.round(234 + (180 - 234) * factor);
        g = Math.round(88 + (160 - 88) * factor);
        b = Math.round(12 + (20 - 12) * factor);
      } else {
        const factor = (ratio - 0.66) / 0.34;
        r = Math.round(180 + (16 - 180) * factor);
        g = Math.round(160 + (185 - 160) * factor);
        b = Math.round(20 + (129 - 20) * factor);
      }

      return {
        color: `rgb(${r}, ${g}, ${b})`,
        shadow: `0 2px 10px rgba(${r}, ${g}, ${b}, 0.2)`
      };
    }

    function initCanvas() {
      const canvas = document.getElementById("sparkleCanvas");
      const container = canvas.parentElement;
      canvas.width = container.offsetWidth + 120;
      canvas.height = container.offsetHeight + 100;
    }

    window.onresize = function() {
      initCanvas();
    };

    function startSparkles(gpa) {
      stopSparkles();
      const canvas = document.getElementById("sparkleCanvas");
      const ctx = canvas.getContext("2d");
      
      const excess = Math.min(1.0, gpa - 4.0);
      const targetSparkleCount = Math.floor(25 + excess * 45);

      sparkles = [];
      for (let i = 0; i < targetSparkleCount; i++) {
        sparkles.push(createSparkle(canvas));
      }

      function animate() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);

        sparkles.forEach((s, index) => {
          s.x += s.vx;
          s.y += s.vy;
          s.alpha += s.dAlpha;

          if (s.alpha >= 1 || s.alpha <= 0) {
            s.dAlpha = -s.dAlpha;
          }

          if (s.y < 0 || s.x < 0 || s.x > canvas.width || s.y > canvas.height) {
            sparkles[index] = createSparkle(canvas);
          }

          ctx.save();
          ctx.translate(s.x, s.y);
          ctx.globalAlpha = Math.max(0, s.alpha);
          ctx.fillStyle = s.color;
          
          ctx.beginPath();
          for (let p = 0; p < 4; p++) {
            ctx.rotate(Math.PI / 2);
            ctx.lineTo(0, s.size);
            ctx.lineTo(s.size * 0.3, s.size * 0.3);
          }
          ctx.fill();
          ctx.restore();
        });

        animationFrameId = requestAnimationFrame(animate);
      }

      animate();
    }

    function stopSparkles() {
      if (animationFrameId) {
        cancelAnimationFrame(animationFrameId);
        animationFrameId = null;
      }
      const canvas = document.getElementById("sparkleCanvas");
      const ctx = canvas.getContext("2d");
      ctx.clearRect(0, 0, canvas.width, canvas.height);
    }

    function createSparkle(canvas) {
      const colors = ["#FBBF24", "#F59E0B", "#FCD34D", "#34D399", "#10B981"];
      return {
        x: Math.random() * canvas.width,
        y: Math.random() * canvas.height,
        vx: (Math.random() - 0.5) * 0.6,
        vy: -Math.random() * 0.8 - 0.2,
        size: Math.random() * 6 + 3,
        alpha: Math.random(),
        dAlpha: (Math.random() * 0.02 + 0.01) * (Math.random() < 0.5 ? 1 : -1),
        color: colors[Math.floor(Math.random() * colors.length)]
      };
    }
